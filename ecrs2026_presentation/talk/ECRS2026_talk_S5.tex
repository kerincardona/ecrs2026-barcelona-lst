% ---------------------------------------------------------------- 2 Problem
\begin{frame}{The Problem: No Baseline, No Way to Measure Change}{Motivation}
  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.9em}
    \item Outdoor thermal exposure depends strongly on urban surface temperature.
    \item The gap: long-term, high-resolution LST climatologies remain scarce for Mediterranean cities.
    \item Without a baseline: cannot measure change, cannot prioritise interventions,
          cannot tell which neighbourhoods are persistently hotter.
    \item Barcelona: 3M+ people in its metropolitan area, a Mediterranean coastal city --
          until now, no systematic, satellite-based thermal baseline.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- 3 Study area
\begin{frame}{Study Area and Data}{Methods -- Data}
  \textfig{%
    \begin{itemize}\setlength{\itemsep}{0.7em}
      \item \textbf{464\,km$^2$} over the Barcelona metropolitan core (21.1 $\times$ 22.0\,km).
      \item \textbf{378} Landsat 8/9 scenes, Collection 2 Level-2, 2013--2024 (12 years).
      \item 30\,m grid, cloud-filtered.
      \item Four seasonal medians (winter, spring, summer, autumn) $\rightarrow$ one climatology.
    \end{itemize}}%
  {fig_studyarea_v2.pdf}
\end{frame}

% ---------------------------------------------------------------- 4 Methods
\begin{frame}{Local Climate Zones + Two Consistency Checks}{Methods -- Classification \& validation}
  \textfig{%
    \small
    \begin{itemize}\setlength{\itemsep}{0.45em}
      \item Local Climate Zones (WUDAPT, Demuzere et al.): building height, density, vegetation cover.
      \item Reference: \textbf{LCZ D} (low plants) -- the rural background.
      \item Two consistency checks:
        \begin{itemize}
          \item ERA5-Land skin temperature, $n=325$ pairs -- seasonal cycle.
          \item ECOSTRESS 70\,m, $n=7$ near-coincident scene pairs ($|\Delta t|\le 2$\,h).
        \end{itemize}
      \item \textbf{Consistency -- not validation.} Seven scene pairs are not enough for full validation.
    \end{itemize}}%
  {fig_lcz_map_col.pdf}
\end{frame}

% ---------------------------------------------------------------- 6 Hotspots
\begin{frame}{Hotspot Analysis: Where Heat Recurs}{Results}
  \textfig{%
    \begin{itemize}\setlength{\itemsep}{0.7em}
      \item Hotspot = pixel above the summer (JJA) P90 threshold.
      \item \textbf{78\%} of the land area (64\% of the window, which includes sea): at least one hotspot event in 12 years.
      \item \textbf{1.3\%} of the land area persistently affected (hotspot frequency $>0.5$) -- the same pixels, year after year.
      \item Where adaptation matters most: tree planting, cool roofs, urban redesign.
    \end{itemize}}%
  {fig_hotspot_col.pdf}
\end{frame}

% ---------------------------------------------------------------- 7 Validation
\begin{frame}{Two Independent Consistency Checks}{Results -- Validation}
  \textfig{%
    \small
    \begin{itemize}\setlength{\itemsep}{0.6em}
      \item \textbf{ERA5-Land (a):} pooled seasonal cycle $r=0.92$.
        \begin{itemize}
          \item Summer (JJA) $r=0.26$: tracks the seasons, not the intra-urban structure (0.1$^\circ$ grid).
        \end{itemize}
      \item \textbf{ECOSTRESS 70\,m (b):} bias $+0.37$\,K, RMSE 4.4\,K, $r=0.92$ -- consistent with the 30\,m field.
      \item ERA5 tells you the season. ECOSTRESS tells you the LST is about right.
            Only Landsat gives the 30\,m structure.
    \end{itemize}}%
  {fig_validation_panel_col.pdf}
\end{frame}

% ---------------------------------------------------------------- 10 Conclusions
\begin{frame}{Three Findings -- and One Takeaway}{Conclusions}
  \vspace{0.3em}
  \begin{enumerate}\setlength{\itemsep}{0.6em}
    \item A 12-year, 30\,m SUHI-by-LCZ climatology for a Mediterranean city -- reproducible, built entirely from open data.
    \item Dense tree cover is 8\,K cooler than the rural reference ($\approx$9.4\,K below large low-rise) --
          vegetation is the dominant spatial cooling signal.
    \item ECOSTRESS confirms our Landsat field is consistent; ERA5 cannot substitute for high-resolution urban LST.
  \end{enumerate}
